## 1. サンプルコードを構成する要素

### C言語っぽい記法とRustっぽい記法
- C言語っぽいところ
  - sizeof を使った境界計算
    - init_heap_area(): size - sizeof(FreeList) * 2
    - malloc(): free_list_addr + sizeof(FreeList) + req_size
    - malloc(): alloc_addr = free_list_addr + sizeof(FreeList)
    - free(): free_list_addr = addr - sizeof(FreeList)
  - 終端・失敗の表現（慣習？）
    - free_list_search(): 終端 free_list_addr == -1 のとき 0 を返す
    - malloc(): 探索失敗で 0 を返す
  - “ヘッダ＋ペイロード”の設計（アロケーター的な記法）
    - malloc(): FreeList を「ヘッダ」として使い、その直後を割当アドレスとして返す
    - free(): addr から sizeof(FreeList) を引いてヘッダに戻る
- Rustっぽいところ
  - 構造体定義
    - struct FreeList { size: i32, prev: daddr, next: daddr }
  - 関数シグネチャ
    - fn free_list_search(... ) -> daddr
    - fn init_heap_area(size: i32)
    - fn malloc(req_size: i32) -> daddr
    - fn free(addr: daddr)
    - fn main()
  - キャスト風記法（アドレス→構造体として解釈）
    - var free_list = free_list_addr as FreeList;
    - (free_list.prev as FreeList).next = ...
    - (free_list.next as FreeList).prev = ...
  - 型注釈
    - req_size: i32 / size: i32 / free_list_addr: daddr / top_addr: daddr
### 連結リスト
- [Wikipedia: 連結リスト](https://ja.wikipedia.org/wiki/%E9%80%A3%E7%B5%90%E3%83%AA%E3%82%B9%E3%83%88)
- ノード構造（prev/next）
  - struct FreeList { prev, next }
- 走査（探索）
  - free_list_search():
    - free_list.size が足りない or 番兵（size==-1）なら free_list.next に進む
- 削除（リストから外す）
  - malloc() の「分割しない」分岐:
    - (free_list.prev as FreeList).next = free_list.next;
    - (free_list.next as FreeList).prev = free_list.prev;
- 挿入（先頭へ繋ぐ）
  - free():
    - free_list.prev = top_addr;
    - free_list.next = (top_addr as FreeList).next;
    - ((top_addr as FreeList).next as FreeList).prev = free_list_addr;
    - (top_addr as FreeList).next = free_list_addr;
- 番兵（sentinel）で先頭管理
  - init_heap_area(): 0x0000 の FreeList を「先頭番兵」として用意（size=-1）
### mallocとfreelist
- [参考記事](https://www.ei.fukui-nct.ac.jp/2018/01/16/malloc-freelist/)
- mallocは動的メモリを確保するための関数
  - 要求サイズに合う空き領域の探索
    - malloc(): free_list_search(0x0000, req_size)
  - 失敗時の戻り値
    - malloc(): 見つからないなら 0
  - 分割（split）
    - malloc(): if (free_list.size > req_size + sizeof(FreeList)) { ... new_free_list を作る }
  - 分割しない（丸ごと確保）
    - malloc(): if (free_list.size <= req_size + sizeof(FreeList)) { ... リストから外す; req_size = free_list.size; }
  - ユーザへ返すのはヘッダ直後
    - malloc(): alloc_addr = free_list_addr + sizeof(FreeList); return alloc_addr;
- freelistは空き領域を管理するためのデータ構造
  - freelist の初期化（番兵＋最初の空きブロック）
    - init_heap_area():
      - 0x0000 に番兵（size=-1, next=0x000c）
      - 0x000c に実 free ブロック（size = ヒープ全体 - ヘッダ2個分）
  - freelist から検索（first-fit 風）
    - free_list_search():
      - 先頭から next を辿り、size >= req_size の最初のブロックを選ぶ
  - freelist から取り出し（split する／しない）
    - malloc(): 見つけた free_list を分割して新しい free を繋ぐ、または丸ごと外す
  - freelist へ戻す（返却されたブロックを先頭へ）
    - free(): 返却ブロックの FreeList を番兵直後に挿入

## 2. 動的メモリ管理アルゴリズムの解説

- サンプルプログラムでは、シンプルな動的メモリ管理アルゴリズムを実装しています。現在商用利用されているものはもっと複雑ですが、基本となるmalloc関数とfree関数の挙動を学んでいきましょう。
- 動的メモリ管理とは、プログラムの実行中に必要に応じてメモリ領域を確保・解放する仕組みです。これにより、プログラムは柔軟にメモリを利用できます。

```
// FreeList: 空きブロックを表す「ヘッダ」構造体
// ヒープ上の「空きブロック」の先頭に置かれる管理情報。
// - size:   このブロックが表す「ユーザに渡せる領域」のサイズ（バイト数想定）
// - prev:   空きリスト上での前ノード（FreeListヘッダ）のアドレス
// - next:   空きリスト上での次ノード（FreeListヘッダ）のアドレス

struct FreeList {
    size: i32,
    prev: daddr,
    next: daddr,
}


// free_list_search: 空き領域リストから「req_size以上」のブロックを探す
// →先頭から順に辿って、最初に見つかった十分なサイズのブロックを返す
//
// 引数:
// - free_list_addr: いま見ている FreeList ヘッダのアドレス
// - req_size:       必要なサイズ
//
// 戻り値:
// - 見つかった FreeList ヘッダのアドレス
// - 見つからないときは 0（失敗）
//
// 実装上の流れ:
// 1) 終端(-1) まで行ったら失敗 0
// 2) free_list_addr を FreeList として読み出す（アドレス→構造体解釈）
// 3) size が小さい、または size == -1（番兵）なら next へ
// 4) size が req_size 以上ならそのブロックを返す
fn free_list_search(free_list_addr: daddr, req_size: i32) -> daddr {
    // -1 は「これ以上ノードが無い」＝リスト終端を表す特別値。
    if (free_list_addr == -1) {
        return 0;
    }

    // 指定アドレスを FreeList 構造体として解釈して参照する
    // （Cでいう: FreeList* free_list = (FreeList*)free_list_addr; みたいな発想）
    var free_list = free_list_addr as FreeList;

    // (1) サイズ不足（free_list.size < req_size）
    // (2) size == -1 は番兵（sentinel）扱い（実データではない）
    // のどちらかなら次のノードへ進む（再帰で探索）
    if (free_list.size < req_size || free_list.size == -1) {
        return free_list_search(free_list.next, req_size);
    }

    return free_list_addr;
}


// init_heap_area: ヒープ領域を初期化し、最初の free list を作る
// 「番兵ノード（常にsize=-1）」を 0x0000 に置き next に「実際の最初の空きブロック」を繋ぐ設計
//
// ヒープ全体サイズ size のうち、最低でも
//   - 番兵FreeListヘッダ
//   - 最初の空きブロックのFreeListヘッダ
// が必要であり、そのぶんを引いた領域が最初の空きサイズになる
fn init_heap_area(size: i32) {
    // 0x0000 : FreeList の先頭 (=ヒープ領域の先頭)
    // 番兵ノード（sentinel）:
    // - size=-1 を「番兵」として扱う
    // - prev=-1 は前が無い（番兵なのでprevは実質使わない）
    // - next=0x000c は「次に実ノードがいる」想定
    //
    // ※ 0x000c は sizeof(FreeList) が 12 bytes の場合の例
    FreeList @ 0x0000 {
        size: -1,
        prev: -1,
        next: 0x000c,
    };
    

    // 0x000c : 実際に管理を行う FreeList の先頭
    // ここが「最初の空きブロック」のヘッダ
    // size は「ヒープ全体」から FreeList ヘッダ2個分を引いたもの
    //
    // 例: size=1024, sizeof(FreeList)=12 の場合:
    //     初期空き = 1024 - 12*2 = 1000 bytes
    //
    // prev は番兵(0x0000)、next は終端(-1)。
    FreeList @ 0x000c {
        size: size - sizeof(FreeList) * 2,
        prev: 0x0000,
        next: -1,
    };
}


// malloc: req_size 分の領域を確保して、ユーザ領域のアドレスを返す
// 1) freelist から req_size を満たす空きブロックを探す
// 2) 見つからないなら 0 を返して失敗
// 3) 見つかった空きブロックが大きければ分割(split)して後半を free として残す
// 4) 小さければ分割せずに丸ごと確保し、リストから取り外す
// 5) 確保したブロックのヘッダ（FreeList）には
//    - size=確保サイズ
//    - prev=0, next=0 といった「確保済み印」を入れる
// 6) ユーザには「ヘッダ直後」のアドレスを返す
//
fn malloc(req_size: i32) -> daddr {
    // 空き領域リストを検索 (失敗した場合は 0 を返す)
    // 検索開始アドレスは番兵(0x0000)
    var free_list_addr = free_list_search(0x0000, req_size);
    if (free_list_addr == 0) {
        return 0;
    }

    // 見つけた空きブロックのヘッダを参照
    var free_list = free_list_addr as FreeList;

    // --------------------------------------------------------
    // 分割(split)の条件:
    // 「要求サイズ + 追加で新ヘッダ(sizeof(FreeList))」を置けるだけの
    // 余りがあるなら split する。
    //
    // つまり:
    //   free_list.size > req_size + sizeof(FreeList)
    //
    // split すると、メモリ配置はこうなる（概念図）:
    //
    // [ free_list_addr ]
    //   +0  : FreeListヘッダ（元の空き）
    //   +H  : ここから req_size が「確保領域」
    //   +H+req_size : ここに新しい FreeListヘッダ（残りの空き）
    //   +H+req_size+H : ここから残りの空き領域
    //
    // H = sizeof(FreeList)
    // --------------------------------------------------------
    if (free_list.size > req_size + sizeof(FreeList)) {
        // 新しい空きブロックのヘッダ位置を計算
        // 「元ヘッダ」 + 「ヘッダ分」 + 「確保分」 の直後が新ヘッダ。
        var new_free_list_addr = free_list_addr + sizeof(FreeList) + req_size;

        // 新しい free ブロックヘッダを書き込む
        // 残りサイズ = 元サイズ - req_size - ヘッダ分
        FreeList @ new_free_list_addr {
            size: free_list.size - req_size - sizeof(FreeList),
            prev: free_list.prev,
            next: free_list.next,
        };

        // ここで「リストのつなぎ替え」を行う。
        // 元の free_list をリストから外して、新しい free_list をそこに入れるイメージ。
        //
        // prev 側の next を new_free_list_addr に更新することで、
        // prev -> (old)free_list だったリンクを prev -> new_free_list に張り替える。
        (free_list.prev as FreeList).next = new_free_list_addr;

        // ★重要な注意（バグになりやすい点）:
        // 双方向リストなら本来ここで「next 側の prev」も new に更新が必要。
        // つまり:
        //   if (free_list.next != -1) (free_list.next as FreeList).prev = new_free_list_addr;
        // が無いと、free_list.next が -1 でないケースでリストが壊れる可能性がある。
        //（今回のテストでは next=-1 が多いなら表面化しにくい）
    }

    // --------------------------------------------------------
    // 分割できない場合:
    // free_list.size が req_size + ヘッダ分 以下なら、
    // split しても「新しい空きブロック」を成立させられない。
    // そこで丸ごと確保し、空きリストからノードを外す。
    // --------------------------------------------------------
    if (free_list.size <= req_size + sizeof(FreeList)) {
        // prev の next を自分の next に飛ばす（自分をスキップ）
        (free_list.prev as FreeList).next = free_list.next;

        // next の prev を自分の prev に戻す（双方向の整合）
        // ※ free_list.next が -1 のときにこれをすると危険。
        //    「-1 as FreeList」を参照してしまう可能性がある。
        //    教育用に簡略化されているが、実装としては終端チェックが必要。
        (free_list.next as FreeList).prev = free_list.prev;

        // 丸ごと確保するので、実際の確保サイズ(req_size)を
        // ブロック全体の size に合わせる（内部的な整合目的）。
        req_size = free_list.size;
    }

    // --------------------------------------------------------
    // 確保ブロックの確定:
    // ユーザへ返すアドレスは「ヘッダ直後」
    // --------------------------------------------------------
    var alloc_addr = free_list_addr + sizeof(FreeList);

    // このヘッダは「もう free list のノードではない」ので、
    // size には確保サイズ、prev/next は 0 を入れて「確保済み印」にする。
    //（0 の意味はこの実装のローカルルール）
    free_list.size = req_size;
    free_list.prev = 0;
    free_list.next = 0;

    return alloc_addr;
}


// ------------------------------------------------------------
// free: malloc で確保した領域を、空き領域リストに戻す
// ------------------------------------------------------------
// 全体の流れ:
// 1) ユーザが持つ addr は「ヘッダ直後」なので、ヘッダ位置に戻す
// 2) そのヘッダを free list の先頭（番兵直後）に挿入する
//
// 注意:
// - 隣接ブロックとの結合(coalescing)が無いので断片化する。
// - 二重解放チェックが無い。
// - ここでも next が -1 だった場合などの安全チェックが弱い。
fn free(addr: daddr) {
    // 返却領域に紐づく FreeList を取得
    // malloc が返したのは "ヘッダ直後" なので、ヘッダに戻るには引く。
    var free_list_addr = addr - sizeof(FreeList);
    var free_list = free_list_addr as FreeList;

    // 空き領域リストの先頭に繋げる
    // ここでは「先頭番兵(top_addr=0x0000)の直後」に挿入する方針。
    var top_addr: daddr = 0x0000;

    // 新ノード（free_list）を番兵の直後に入れるので、
    // free_list.prev = 番兵
    free_list.prev = top_addr;

    // free_list.next = 番兵の現在の next（元の先頭ノード）
    free_list.next = (top_addr as FreeList).next;

    // 元の先頭ノードの prev を free_list_addr に更新
    // ※ 番兵.next が -1 の場合（空きリストが空）には危険。
    //    その場合は終端チェックして分岐が必要。
    ((top_addr as FreeList).next as FreeList).prev = free_list_addr;

    // 番兵の next を新ノードに更新して挿入完了
    (top_addr as FreeList).next = free_list_addr;
}


// main: 動作例（簡易テスト）
// - 1024 bytes のヒープを初期化
// - 100, 200, 50 bytes を順に確保
// - 200 のブロックだけ free
// - 150 を確保（first-fitで 200 の空きから取る想定）
//
// 期待される概念的な挙動:
// - ptr2 を free すると、そのブロックが freelist に戻る
// - ptr4=malloc(150) は、戻した 200 のブロックを利用し、
//   余りがあれば split して残り 200-150-ヘッダ分 が free として残る
fn main() {
    init_heap_area(1024);

    var ptr1 = malloc(100);
    var ptr2 = malloc(200);
    var ptr3 = malloc(50);

    free(ptr2);
    var ptr4 = malloc(150);
}
```
